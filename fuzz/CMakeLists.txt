if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "BUILD_FUZZING requires a Clang toolchain with libFuzzer")
endif()
if(NOT ENABLE_SANITIZERS)
    message(FATAL_ERROR "BUILD_FUZZING requires ENABLE_SANITIZERS=ON")
endif()

add_executable(fuzz_json json.cpp)
target_link_libraries(fuzz_json PRIVATE slimr::json slimr_options)
add_executable(fuzz_yaml yaml.cpp)
target_link_libraries(fuzz_yaml PRIVATE slimr::yaml slimr_options)
add_executable(fuzz_encoding encoding.cpp)
target_link_libraries(fuzz_encoding PRIVATE slimr::encoding slimr_options)
foreach(target fuzz_json fuzz_yaml fuzz_encoding)
    target_compile_options(${target} PRIVATE -O1 -g -fsanitize=fuzzer)
    target_link_options(${target} PRIVATE -fsanitize=fuzzer)
endforeach()
